cmake_minimum_required(VERSION 3.20)
if(NOT DEFINED EFF_PROJECT_ROOT)
    project(EfficiencyCommon LANGUAGES CXX)
    include(CTest)
    set(EFF_BUILD_SIMULATION OFF CACHE BOOL "" FORCE)
    set(EFF_BUILD_MERGER OFF CACHE BOOL "" FORCE)
    add_subdirectory(.. suite)
    return()
endif()
add_library(eff_common src/SimulationConfig.cpp src/TaskSpec.cpp src/CellCounts.cpp)
target_include_directories(eff_common PUBLIC include "${EFF_PROJECT_ROOT}/third_party")
target_compile_options(eff_common PRIVATE -Wall -Wextra -Wpedantic)
if(EFF_BUILD_SIMULATION OR EFF_BUILD_MERGER)
    add_library(eff_root_io io/RootCountsIO.cpp)
    # TTreeReader 的实现属于 TreePlayer，不属于 Tree。PUBLIC 让模拟器、合并器、
    # ROOT 往返测试统一继承相同的库和 C++ 运行库，不能逐个可执行文件补丁式链接。
    target_link_libraries(eff_root_io PUBLIC eff_common ROOT::Core ROOT::RIO ROOT::Tree ROOT::TreePlayer eff_runtime)
endif()
if(BUILD_TESTING)
    add_executable(test_counts "${EFF_PROJECT_ROOT}/tests/test_counts.cpp")
    target_link_libraries(test_counts PRIVATE eff_common)
    add_test(NAME counts COMMAND test_counts "${EFF_PROJECT_ROOT}/tests/fixtures/sim_config.json")
    if(EFF_BUILD_SIMULATION OR EFF_BUILD_MERGER)
        add_executable(test_root_io "${EFF_PROJECT_ROOT}/tests/test_root_io.cpp")
        target_link_libraries(test_root_io PRIVATE eff_root_io)
        add_test(NAME root_io COMMAND test_root_io "${EFF_PROJECT_ROOT}/tests/fixtures/sim_config.json")
    endif()
endif()
