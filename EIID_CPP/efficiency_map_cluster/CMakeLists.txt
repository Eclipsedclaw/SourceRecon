cmake_minimum_required(VERSION 3.20)
project(EfficiencyMapCluster VERSION 1.0 LANGUAGES CXX)
set(EFF_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
option(EFF_BUILD_SIMULATION "Build the headless Geant4 MT simulator" ON)
option(EFF_BUILD_MERGER "Build the ROOT count merger" ON)
set(CMAKE_CXX_STANDARD 17 CACHE STRING "C++ standard (raised if ROOT/Geant4 require it)")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(CTest)
file(REMOVE "${CMAKE_BINARY_DIR}/configure.ok")
include(cmake/DiscoverDependencies.cmake)

if(EFF_BUILD_SIMULATION OR EFF_BUILD_MERGER)
    find_package(ROOT 6.28 REQUIRED COMPONENTS Core RIO Tree TreePlayer)
    if(DEFINED ROOT_CXX_STANDARD AND ROOT_CXX_STANDARD GREATER CMAKE_CXX_STANDARD)
        set(CMAKE_CXX_STANDARD "${ROOT_CXX_STANDARD}")
    endif()
    if(NOT EFF_ROOT_CONFIG)
        find_program(EFF_ROOT_CONFIG root-config HINTS "${ROOT_BINDIR}" "${ROOT_DIR}/../../../bin" "${ROOT_DIR}/../../bin" NO_DEFAULT_PATH REQUIRED)
    endif()
    execute_process(COMMAND "${EFF_ROOT_CONFIG}" --cflags OUTPUT_VARIABLE ROOT_FLAGS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(ROOT_FLAGS MATCHES "-std=c\\+\\+([0-9]+)")
        if(CMAKE_MATCH_1 GREATER CMAKE_CXX_STANDARD)
            set(CMAKE_CXX_STANDARD "${CMAKE_MATCH_1}")
        endif()
    endif()
endif()

if(EFF_BUILD_SIMULATION)
    # 不请求 ui_all/vis_all；不能把串行 Geant4 当多线程库使用。
    find_package(Geant4 11.2 REQUIRED COMPONENTS multithreaded)
    if(NOT TARGET Geant4::G4run)
        message(FATAL_ERROR "This runtime wrapper requires a shared-library Geant4 MT installation (Geant4::G4run).")
    endif()
    if(DEFINED Geant4_CXX_STANDARD AND Geant4_CXX_STANDARD GREATER CMAKE_CXX_STANDARD)
        set(CMAKE_CXX_STANDARD "${Geant4_CXX_STANDARD}")
    endif()
    find_path(HEALPIX_INCLUDE_DIR healpix_base.h HINTS "$ENV{CONDA_PREFIX}/include" PATH_SUFFIXES healpix_cxx)
    find_library(HEALPIX_LIBRARY NAMES healpix_cxx HINTS "$ENV{CONDA_PREFIX}/lib")
    find_library(CXXSUPPORT_LIBRARY NAMES cxxsupport HINTS "$ENV{CONDA_PREFIX}/lib")
    if(NOT HEALPIX_INCLUDE_DIR OR NOT HEALPIX_LIBRARY)
        message(FATAL_ERROR "HEALPix missing. Set CMAKE_PREFIX_PATH or HEALPIX_INCLUDE_DIR/HEALPIX_LIBRARY.")
    endif()
    add_library(eff_healpix UNKNOWN IMPORTED)
    set_target_properties(eff_healpix PROPERTIES IMPORTED_LOCATION "${HEALPIX_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${HEALPIX_INCLUDE_DIR}")
    if(CXXSUPPORT_LIBRARY)
        set_property(TARGET eff_healpix APPEND PROPERTY INTERFACE_LINK_LIBRARIES "${CXXSUPPORT_LIBRARY}")
    endif()
    # 一些静态 HEALPix 构建启用了 OpenMP；它是库依赖，不用于并行 Geant4。
    find_package(OpenMP QUIET)
    if(OpenMP_CXX_FOUND)
        set_property(TARGET eff_healpix APPEND PROPERTY INTERFACE_LINK_LIBRARIES OpenMP::OpenMP_CXX)
    endif()
    find_program(EFF_GEANT4_CONFIG geant4-config HINTS "${Geant4_DIR}/../../../bin" "${Geant4_DIR}/../../bin" NO_DEFAULT_PATH REQUIRED)
    execute_process(COMMAND "${EFF_GEANT4_CONFIG}" --version OUTPUT_VARIABLE EFF_G4_CONFIG_VERSION
        OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(NOT EFF_G4_CONFIG_VERSION VERSION_EQUAL Geant4_VERSION)
        message(FATAL_ERROR "geant4-config and Geant4Config.cmake select different versions. Set EFF_GEANT4_CONFIG explicitly.")
    endif()
    get_filename_component(EFF_GEANT4_BIN "${EFF_GEANT4_CONFIG}" DIRECTORY)
    set(EFF_GEANT4_SH "${EFF_GEANT4_BIN}/geant4.sh")
    if(NOT EXISTS "${EFF_GEANT4_SH}")
        message(FATAL_ERROR "Cannot locate the matching geant4.sh beside geant4-config")
    endif()
endif()

# 配置时实际链接并启动内存 ROOT/HEALPix/Geant4 小检查，不运行输运、不写用户数据。
include(cmake/CheckDependencies.cmake)

# 文件变更会触发 CMake 更新指纹，防止同一批任务混入不同版本的物理代码。
file(GLOB_RECURSE EFF_SOURCES CONFIGURE_DEPENDS common/*.h common/*.cpp Simulation/*.hh Simulation/*.cc Simulation/*.cpp)
list(APPEND EFF_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/third_party/json.hpp")
set(EFF_DIGEST "")
foreach(SOURCE IN LISTS EFF_SOURCES)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SOURCE}")
    file(SHA256 "${SOURCE}" HASH)
    string(APPEND EFF_DIGEST "${HASH}")
endforeach()
string(SHA256 EFF_SOURCE_SHA256 "${EFF_DIGEST}")
set(EFF_COMPILER "${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}-c++${CMAKE_CXX_STANDARD}")
set(EFF_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
configure_file(cmake/BuildInfo.h.in "${EFF_GENERATED_DIR}/BuildInfo.h" @ONLY)
add_subdirectory(common)
if(EFF_BUILD_SIMULATION)
    add_subdirectory(Simulation)
endif()
if(EFF_BUILD_MERGER)
    add_subdirectory(Merger)
endif()

if(EFF_BUILD_SIMULATION)
    get_filename_component(EFF_HEALPIX_LIBDIR "${HEALPIX_LIBRARY}" DIRECTORY)
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/runtime_paths.sh" CONTENT
        "export EFF_GEANT4_SH='${EFF_GEANT4_SH}'\nexport EFF_LIBRARY_PATH='${EFF_RUNTIME_PATH}'\n")
elseif(EFF_BUILD_MERGER)
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/runtime_paths.sh" CONTENT
        "export EFF_GEANT4_SH=''\nexport EFF_LIBRARY_PATH='${EFF_RUNTIME_PATH}'\n")
else()
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/runtime_paths.sh" CONTENT
        "export EFF_GEANT4_SH=''\nexport EFF_LIBRARY_PATH=''\n")
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/configure.ok" CONTENT "efficiency-map-cluster-build-check-v2\n")
message(STATUS "Efficiency cluster: ${EFF_COMPILER}; headless, no plotting or particle transport during configure")
